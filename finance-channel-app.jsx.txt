// ─── Claude API ───────────────────────────────────────────────────────────────
async function callClaude(systemPrompt, userPrompt) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-20250514",
      max_tokens: 1000,
      system: systemPrompt,
      messages: [{ role: "user", content: userPrompt }],
    }),
  });
  const data = await res.json();
  return data.content?.[0]?.text || "";
}

function Schedule({ videos, setVideos }) {
  const today = new Date();
  const year = today.getFullYear();
  const month = today.getMonth();
  const firstDay = new Date(year, month, 1).getDay();
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const monthName = today.toLocaleString("default", { month: "long" });

  const videosByDate = {};
  videos.forEach(v => { videosByDate[v.date] = v; });

  const cells = [];
  for (let i = 0; i < firstDay; i++) cells.push(null);
  for (let d = 1; d <= daysInMonth; d++) cells.push(d);

  const dateStr = (d) => `${year}-${String(month + 1).padStart(2, "0")}-${String(d).padStart(2, "0")}`;
  const isPostDay = (d) => POST_DAYS.includes(new Date(year, month, d).getDay());
  const isToday = (d) => d === today.getDate();

  const [newTitle, setNewTitle] = useState("");
  const [newDate, setNewDate] = useState("");
  const [newType, setNewType] = useState("long");

  const addVideo = () => {
    if (!newTitle || !newDate) return;
    setVideos(v => [...v, { id: Date.now(), title: newTitle, status: "scheduled", date: newDate, views: 0, type: newType }]);
    setNewTitle(""); setNewDate("");
  };

  return (
    <div>
      <div className="page-title">Content Schedule</div>
      <div className="page-sub">Wed · Fri · Sun posting cadence</div>

      <div className="grid-2 section-gap">
        <div className="card">
          <div className="card-title">{monthName} {year}</div>
          <div className="cal-grid mb-12">
            {DAYS.map(d => <div key={d} className="cal-header">{d}</div>)}
            {cells.map((d, i) => (
              <div key={i} className={`cal-day ${!d ? "empty" : ""} ${d && isPostDay(d) ? "post-day" : ""} ${d && isToday(d) ? "today" : ""} ${d && videosByDate[dateStr(d)] ? "has-video" : ""}`}>
                {d}
              </div>
            ))}
          </div>
          <div className="flex gap-8" style={{ flexWrap: "wrap" }}>
            <span style={{ fontSize: 11, color: COLORS.muted }}>
              <span style={{ display: "inline-block", width: 10, height: 10, borderRadius: 2, background: COLORS.accentDim, marginRight: 4 }}></span>Post day
            </span>
            <span style={{ fontSize: 11, color: COLORS.muted }}>
              <span style={{ display: "inline-block", width: 8, height: 8, borderRadius: "50%", background: COLORS.gold, marginRight: 4 }}></span>Video scheduled
            </span>
          </div>
        </div>

        <div className="card">
          <div className="card-title">All Videos</div>
          <div style={{ maxHeight: 320, overflowY: "auto" }}>
            {videos.map(v => (
              <div className="video-row" key={v.id}>
                <div style={{ flex: 1 }}>
                  <div className="video-title mb-4">{v.title}</div>
                  <div className="flex gap-8">
                    <span className={`badge badge-${v.status}`}>{v.status}</span>
                    {v.type === "short" && <span className="badge badge-short">short</span>}
                    <span className="video-date">{v.date}</span>
                  </div>
                </div>
              </div>
            ))}
          </div>
        </div>
      </div>

      <div className="card">
        <div className="card-title">Add Video</div>
        <div className="flex gap-8" style={{ flexWrap: "wrap" }}>
          <input className="input" placeholder="Video title" value={newTitle} onChange={e => setNewTitle(e.target.value)} style={{ flex: 2, minWidth: 200 }} />
          <input className="input" type="date" value={newDate} onChange={e => setNewDate(e.target.value)} style={{ flex: 1, minWidth: 140 }} />
          <select className="input" value={newType} onChange={e => setNewType(e.target.value)} style={{ flex: 1, minWidth: 100 }}>
            <option value="long">Long-form</option>
            <option value="short">Short</option>

        </div>
      </div>
    </div>
  );
}

function Script() {
  const [title, setTitle] = useState("");
  const [duration, setDuration] = useState("8");
  const [tab, setTab] = useState("outline");
  const [output, setOutput] = useState({ outline: "", hook: "", script: "" });
  const [loading, setLoading] = useState(false);

  const generate = async (type) => {
    if (!title) return;
    setLoading(true);
    setTab(type);
    try {
      const sys = `You are a YouTube scriptwriter for a faceless finance channel run by a Chartered Accountant. The host never appears on camera. Content is for educated, salaried Indians aged 25-40. Use simple but authoritative language. Leverage the CA credential for credibility.`;

      const prompts = {
        outline: `Create a detailed script outline for a ${duration}-minute YouTube video titled: "${title}". Include: intro hook idea, 4-5 main sections with key points, and a CTA. Format clearly with section headers.`,
        hook: `Write 3 different opening hooks (first 30 seconds) for a YouTube video titled: "${title}". Each hook should use a different technique: 1) Bold statement / stat, 2) Relatable pain point, 3) Curiosity gap. Keep each under 60 words.`,
        script: `Write a full script for a ${duration}-minute YouTube video titled: "${title}". Include: [HOOK], [MAIN CONTENT with clear transitions], [CTA]. Use conversational but authoritative tone. Add [B-ROLL: description] cues. Keep it tight and punchy.`,
      };

      const res = await callClaude(sys, prompts[type]);
      setOutput(o => ({ ...o, [type]: res }));
    } catch {
      setOutput(o => ({ ...o, [type]: "Error. Please try again." }));
    }
    setLoading(false);
  };

  return (
    <div>
      <div className="page-title">AI Script Writer</div>
      <div className="page-sub">From idea to full script in seconds</div>

      <div className="card section-gap">
        <div className="card-title">Video Details</div>
        <div className="flex gap-8 mb-16" style={{ flexWrap: "wrap" }}>
          <input className="input" placeholder="Video title or topic..." value={title} onChange={e => setTitle(e.target.value)} style={{ flex: 3, minWidth: 200 }} />
          <select className="input" value={duration} onChange={e => setDuration(e.target.value)} style={{ flex: 1, minWidth: 100 }}>
            {["5", "8", "10", "12", "15"].map(d => <option key={d} value={d}>{d} mins</option>)}
          </select>
        </div>

        <div className="tabs">
          {["outline", "hook", "script"].map(t => (
            <button key={t} className={`tab ${tab === t ? "active" : ""}`} onClick={() => setTab(t)}>
              {t === "outline" ? "◈ Outline" : t === "hook" ? "⚡ Hook" : "◎ Full Script"}
            </button>
          ))}
        </div>

        <div className="flex gap-8 mb-12">
          <button className="btn btn-primary btn-sm" onClick={() => generate(tab)} disabled={loading || !title}>
            {loading ? "Writing..." : `Generate ${tab}`}
          </button>
          <button className="btn btn-ghost btn-sm" onClick={() => generate("outline")} disabled={loading || !title}>Outline</button>
          <button className="btn btn-ghost btn-sm" onClick={() => generate("hook")} disabled={loading || !title}>Hooks</button>
          <button className="btn btn-ghost btn-sm" onClick={() => generate("script")} disabled={loading || !title}>Full Script</button>
        </div>

        <div className={`ai-output ${loading ? "ai-loading" : ""}`} style={{ maxHeight: 400, overflowY: "auto" }}>
          {loading ? "⬡ Writing your script..." : output[tab] || `Click "Generate ${tab}" to get started...`}
        </div>
      </div>
    </div>
  );
}

function Checklist() {
  const ITEMS = [
    { id: 1, phase: "Pre-Production", text: "Topic researched & angle confirmed" },
    { id: 2, phase: "Pre-Production", text: "Script written & reviewed" },
    { id: 3, phase: "Pre-Production", text: "Thumbnail concept decided" },
    { id: 4, phase: "Production", text: "Voiceover recorded (clean audio)" },
    { id: 5, phase: "Production", text: "B-roll / screen recordings done" },
    { id: 6, phase: "Post-Production", text: "Video edited & exported" },
    { id: 7, phase: "Post-Production", text: "Thumbnail designed" },
    { id: 8, phase: "Upload", text: "Title optimised with keywords" },
    { id: 9, phase: "Upload", text: "Description written (with timestamps)" },
    { id: 10, phase: "Upload", text: "Tags added (10-15 relevant tags)" },
    { id: 11, phase: "Upload", text: "End screen & cards added" },
    { id: 12, phase: "Upload", text: "Shorts version cut & scheduled" },
    { id: 13, phase: "Post-Upload", text: "Pinned comment added" },
    { id: 14, phase: "Post-Upload", text: "Shared to communities / socials" },
  ];
  const phases = [...new Set(ITEMS.map(i => i.phase))];
  const [checked, setChecked] = useState({});

  const toggle = id => setChecked(c => ({ ...c, [id]: !c[id] }));
  const done = Object.values(checked).filter(Boolean).length;
  const reset = () => setChecked({});

  return (
    <div>
      <div className="page-title">Pre-Publish Checklist</div>
      <div className="page-sub">Every video, every time</div>

      <div className="card section-gap">
        <div className="flex-between mb-16">
          <div>
            <span style={{ fontSize: 28, fontWeight: 800, color: COLORS.accent }}>{done}</span>
            <span style={{ fontSize: 16, color: COLORS.muted }}> / {ITEMS.length} complete</span>
          </div>
          <button className="btn btn-ghost btn-sm" onClick={reset}>Reset</button>
        </div>
        <div style={{ height: 4, background: COLORS.border, borderRadius: 2, marginBottom: 20, overflow: "hidden" }}>
          <div style={{ height: "100%", width: `${(done / ITEMS.length) * 100}%`, background: COLORS.accent, borderRadius: 2, transition: "width 0.3s" }}></div>
        </div>

        {phases.map(phase => (
          <div key={phase} className="mb-16">
            <div style={{ fontSize: 11, fontWeight: 700, color: COLORS.gold, letterSpacing: 1, textTransform: "uppercase", marginBottom: 8, fontFamily: "'DM Mono', monospace" }}>
              {phase}
            </div>
            {ITEMS.filter(i => i.phase === phase).map(item => (
              <div key={item.id} className="check-item" onClick={() => toggle(item.id)} style={{ cursor: "pointer" }}>
                <div className={`check-box ${checked[item.id] ? "checked" : ""}`}>
                  {checked[item.id] && "✓"}
                </div>
                <span className={`check-text ${checked[item.id] ? "done" : ""}`}>{item.text}</span>
              </div>
            ))}
          </div>
        ))}
      </div>
    </div>
  );
}
